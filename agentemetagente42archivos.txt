7. src/core/eventBus.js
javascript
export class EventBus {
  constructor() {
    this.events = new Map();
  }

  on(event, callback) {
    if (!this.events.has(event)) {
      this.events.set(event, new Set());
    }
    this.events.get(event).add(callback);
    return () => this.off(event, callback);
  }

  off(event, callback) {
    if (this.events.has(event)) {
      this.events.get(event).delete(callback);
    }
  }

  emit(event, payload) {
    if (this.events.has(event)) {
      this.events.get(event).forEach(cb => {
        try {
          cb(payload);
        } catch (err) {
          console.error(`Error in event ${event}:`, err);
        }
      });
    }
  }
}

import { EVENTS } from '../utils/constants.js';

export class AgentRegistry {
  #agents = new Map();
  #eventBus;

  constructor(eventBus) {
    this.#eventBus = eventBus;
  }

  register(config) {
    const id = config.id || `${config.name.toLowerCase().replace(/\s+/g, '-')}-${Date.now()}`;
    const agentConfig = { ...config, id };
    this.#agents.set(id, agentConfig);
    this.#eventBus.emit(EVENTS.AGENT_REGISTERED, { agentConfig });
    return id;
  }

  update(id, config) {
    if (!this.#agents.has(id)) return false;
    const updated = { ...config, id };
    this.#agents.set(id, updated);
    this.#eventBus.emit(EVENTS.AGENT_UPDATED, { agentId: id, config: updated });
    return true;
  }

  remove(id) {
    if (!this.#agents.has(id)) return false;
    this.#agents.delete(id);
    this.#eventBus.emit(EVENTS.AGENT_REMOVED, { agentId: id });
    return true;
  }

  get(id) {
    return this.#agents.get(id);
  }

  list() {
    return Array.from(this.#agents.values());
  }
}

import { EVENTS } from '../utils/constants.js';

export class Orchestrator {
  #eventBus;
  #registry;
  #context;

  constructor(eventBus, registry, context) {
    this.#eventBus = eventBus;
    this.#registry = registry;
    this.#context = context;

    this.#eventBus.on(EVENTS.USER_COMMAND, this.#handleUserCommand.bind(this));
    this.#eventBus.on(EVENTS.NLP_RESULT, this.#handleNLPResult.bind(this));
  }

  #handleUserCommand({ rawText }) {
    this.#eventBus.emit(EVENTS.NLP_ANALYSIS_REQUESTED, { text: rawText });
  }

  #handleNLPResult({ intent, entities }) {
    const agent = this.#selectAgent(intent);
    if (!agent) {
      this.#eventBus.emit(EVENTS.ORCHESTRATOR_RESPONSE, {
        message: 'No tengo un agente para esa petición.'
      });
      return;
    }

    this.#eventBus.emit(EVENTS.AGENT_DELEGATE, {
      agentId: agent.id,
      params: entities
    });
  }

  #selectAgent(intent) {
    const agents = this.#registry.list();
    return agents.find(agent => agent.keywords?.some(kw => intent.includes(kw)));
  }
}

11. src/core/pipelineExecutor.js
javascript
import { InputStep } from '../steps/inputStep.js';
import { FilterStep } from '../steps/filterStep.js';
import { TransformStep } from '../steps/transformStep.js';
import { OutputStep } from '../steps/outputStep.js';
import { ConditionStep } from '../steps/conditionStep.js';
import { LoopStep } from '../steps/loopStep.js';
import { CodeCheckStep } from '../steps/codeCheckStep.js';
import { CallApiStep } from '../steps/callApiStep.js';

export class PipelineExecutor {
  static stepMap = {
    input: InputStep,
    filter: FilterStep,
    transform: TransformStep,
    output: OutputStep,
    condition: ConditionStep,
    loop: LoopStep,
    codeCheck: CodeCheckStep,
    callApi: CallApiStep
  };

  constructor(eventBus, context, agentId) {
    this.eventBus = eventBus;
    this.context = context;
    this.agentId = agentId;
  }

  async execute(pipeline, initialParams = {}) {
    let localContext = { ...this.context.getAll(), ...initialParams };
    const steps = pipeline;

    for (let i = 0; i < steps.length; i++) {
      const stepConfig = steps[i];
      const StepClass = PipelineExecutor.stepMap[stepConfig.type];
      if (!StepClass) {
        throw new Error(`Tipo de paso desconocido: ${stepConfig.type}`);
      }

      const stepInstance = new StepClass(this.eventBus, this.agentId);
      try {
        localContext = await stepInstance.execute(localContext, stepConfig);
      } catch (error) {
        this.eventBus.emit('pipeline:stepError', { agentId: this.agentId, step: stepConfig, error });
        throw error;
      }

      if (localContext.__halt) {
        break;
      }
    }

    return localContext;
  }
}

export class BaseStep {
  constructor(eventBus, agentId) {
    this.eventBus = eventBus;
    this.agentId = agentId;
  }

  async execute(context, config) {
    throw new Error('execute() must be implemented');
  }

  evaluateExpression(expression, context) {
    const func = new Function('context', `return (${expression})`);
    return func(context);
  }

  interpolate(template, context) {
    return template.replace(/\{\{([^}]+)\}\}/g, (_, path) => {
      return path.trim().split('.').reduce((obj, key) => obj?.[key], context) ?? '';
    });
  }
}

import { BaseStep } from './baseStep.js';

export class InputStep extends BaseStep {
  async execute(context, config) {
    this.eventBus.emit('pipeline:inputRequest', {
      agentId: this.agentId,
      fields: config.fields,
      callback: (userInput) => {
        this.eventBus.emit('pipeline:inputReceived', { agentId: this.agentId, input: userInput });
      }
    });

    return new Promise((resolve) => {
      const handler = (data) => {
        if (data.agentId === this.agentId) {
          this.eventBus.off('pipeline:inputReceived', handler);
          resolve({ ...context, input: data.input });
        }
      };
      this.eventBus.on('pipeline:inputReceived', handler);
    });
  }
}

import { BaseStep } from './baseStep.js';
import { CustomJsonAdapter } from '../dataAdapters/customJsonAdapter.js';
import { applyDeclarativeRules } from '../rulesEngine/ruleEngine.js';

export class FilterStep extends BaseStep {
  async execute(context, config) {
    const adapter = new CustomJsonAdapter();
    const dataset = await adapter.fetch({ customDataset: config.dataset });

    const userInput = {};
    for (const rule of config.rules) {
      userInput[rule.field] = this.interpolate(rule.value, context);
    }

    const filtered = applyDeclarativeRules(dataset, { dimensions: { goal: { rules: { '*': config.rules[0] } } } }, userInput);
    context.filtered = filtered;
    return context;
  }
}

export class ConditionStep extends BaseStep {
  async execute(context, config) {
    const condition = this.evaluateExpression(config.expression, context);
    const subPipeline = condition ? config.truePipeline : config.falsePipeline;
    if (subPipeline) {
      const executor = new PipelineExecutor(this.eventBus, { getAll: () => context }, this.agentId);
      return await executor.execute(subPipeline, context);
    }
    return context;
  }
}

export class LoopStep extends BaseStep {
  async execute(context, config) {
    const items = context[config.itemsField] || [];
    const results = [];
    const executor = new PipelineExecutor(this.eventBus, { getAll: () => context }, this.agentId);
    
    for (const item of items) {
      const itemContext = { ...context, item };
      const result = await executor.execute(config.pipeline, itemContext);
      results.push(result);
    }
    
    context[config.outputField || 'loopResults'] = results;
    return context;
  }
}

import { BaseStep } from './baseStep.js';

export class CallApiStep extends BaseStep {
  async execute(context, config) {
    const url = this.interpolate(config.url, context);
    const options = {
      method: config.method || 'GET',
      headers: config.headers || {}
    };

    if (config.bodyTemplate) {
      options.body = JSON.stringify(this.interpolate(config.bodyTemplate, context));
    }

    const response = await fetch(url, options);
    const data = await response.json();
    context[config.outputField || 'apiResponse'] = data;
    return context;
  }
}

import { EVENTS } from '../utils/constants.js';

export class UserPanel {
  #eventBus;
  #container;
  #currentInputCallback = null;

  constructor(eventBus) {
    this.#eventBus = eventBus;
    this.#container = document.createElement('div');
    this.#container.className = 'card user-panel';
    this.#eventBus.on(EVENTS.ORCHESTRATOR_RESPONSE, this.#showResponse.bind(this));
    this.#eventBus.on('pipeline:inputRequest', this.#handleInputRequest.bind(this));
    this.#eventBus.on('pipeline:output', this.#showOutput.bind(this));
  }

  render() {
    this.#container.innerHTML = `
      <h2>🤖 Asistente</h2>
      <label>Escribe tu petición:</label>
      <textarea id="userCommand" rows="2" placeholder="Ej: Quiero una receta light en 10 minutos"></textarea>
      <button class="btn" id="sendCommandBtn">🚀 Enviar</button>
      <div id="dynamicInputArea"></div>
      <div id="agentOutput" class="preview-box">Esperando tu orden...</div>
    `;

    this.#container.querySelector('#sendCommandBtn').addEventListener('click', () => {
      const text = this.#container.querySelector('#userCommand').value;
      this.#eventBus.emit(EVENTS.USER_COMMAND, { rawText: text });
    });

    return this.#container;
  }

  #showResponse({ message }) {
    this.#container.querySelector('#agentOutput').innerHTML = message;
  }

  #showOutput({ value }) {
    this.#container.querySelector('#agentOutput').innerHTML = value;
  }

  #handleInputRequest({ fields, callback }) {
    this.#currentInputCallback = callback;
    const area = this.#container.querySelector('#dynamicInputArea');
    let html = '<div class="input-modal"><h3>Completa los datos</h3>';
    fields.forEach(field => {
      html += `<label>${field}</label><input id="input_${field}" placeholder="${field}" />`;
    });
    html += '<button class="btn" id="submitInputBtn">Enviar</button></div>';
    area.innerHTML = html;

    area.querySelector('#submitInputBtn').addEventListener('click', () => {
      const input = {};
      fields.forEach(field => {
        input[field] = area.querySelector(`#input_${field}`).value;
      });
      area.innerHTML = '';
      this.#currentInputCallback(input);
    });
  }
}

    this.registry = registry;
    this.eventBus.on(EVENTS.AGENT_REGISTERED, () => this.save());
    this.eventBus.on(EVENTS.AGENT_UPDATED, () => this.save());
    this.eventBus.on(EVENTS.AGENT_REMOVED, () => this.save());
  }

  load() {
    const saved = localStorage.getItem('agentsConfig');
    if (saved) {
      const configs = JSON.parse(saved);
      configs.forEach(cfg => this.registry.register(cfg));
    }
  }

  save() {
    const configs = this.registry.list();
    localStorage.setItem('agentsConfig', JSON.stringify(configs));
  }
}

export class EntityExtractor {
  extract(text) {
    const entities = {};
    const timeMatch = text.match(/(\d+)\s*(min|minutos?)/i);
    if (timeMatch) entities.maxTime = parseInt(timeMatch[1]);
    if (text.includes('light')) entities.goal = 'light';
    if (text.includes('proteína')) entities.goal = 'protein';
    return entities;
  }
}

import { EVENTS } from '../utils/constants.js';
import { IntentClassifier } from './intentClassifier.js';
import { EntityExtractor } from './entityExtractor.js';

export class NLPipeline {
  #classifier;
  #extractor;
  #eventBus;

  constructor(eventBus) {
    this.#classifier = new IntentClassifier();
    this.#extractor = new EntityExtractor();
    this.#eventBus = eventBus;
    this.#eventBus.on(EVENTS.NLP_ANALYSIS_REQUESTED, this.#analyze.bind(this));
  }

  #analyze({ text }) {
    const intent = this.#classifier.classify(text);
    const entities = this.#extractor.extract(text);
    this.#eventBus.emit(EVENTS.NLP_RESULT, { intent, entities, confidence: 0.8 });
  }
}

export function applyDeclarativeRules(items, config, userInput) {
  let filtered = [...items];
  // Implementación simplificada para el ejemplo
  if (config.dimensions?.goal?.rules) {
    const rules = config.dimensions.goal.rules['*'];
    if (rules) {
      filtered = filtered.filter(item => {
        const itemValue = item[rules.campo];
        const userValue = userInput[rules.campo];
        switch (rules.operator) {
          case '=': return itemValue == userValue;
          case '<=': return itemValue <= userValue;
          default: return true;
        }
      });
    }
  }
  return filtered;
}
